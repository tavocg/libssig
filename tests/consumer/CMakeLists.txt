cmake_minimum_required(VERSION 3.16)
project(libssig_consumer LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
if(LIBSSIG_SOURCE)
    # An embedded library must not enable its own tests/install rules or change
    # the parent's shared-library preference.
    set(parent_shared "${BUILD_SHARED_LIBS}")
    add_subdirectory("${LIBSSIG_SOURCE}" libssig)
    if(LIBSSIG_BUILD_TESTING OR LIBSSIG_INSTALL OR TARGET noop_test)
        message(FATAL_ERROR "Embedded libssig enabled tests or installation")
    endif()
    if(NOT "${BUILD_SHARED_LIBS}" STREQUAL "${parent_shared}")
        message(FATAL_ERROR "Embedded libssig changed BUILD_SHARED_LIBS")
    endif()
else()
    find_package(libssig 0.1.5 CONFIG REQUIRED)
endif()
get_target_property(actual_type libssig::ssig TYPE)
if(NOT actual_type STREQUAL EXPECTED_TYPE)
    message(FATAL_ERROR "Expected ${EXPECTED_TYPE}, got ${actual_type}")
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE libssig::ssig)
add_test(NAME consumer COMMAND consumer)
